# GPU runtime with PyTorch preinstalled (CUDA 12.8)
FROM pytorch/pytorch:2.9.0-cuda12.8-cudnn9-runtime

ENV PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1 \
    PIP_NO_CACHE_DIR=1

WORKDIR /app

# --- System deps ---
# Tkinter GUI + X11 libs
# OpenCV runtime libs
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential git curl ca-certificates \
    python3-tk tk tcl \
    libx11-6 libxext6 libxrender1 libsm6 \
    libgl1 libglib2.0-0 \
    ffmpeg \
 && rm -rf /var/lib/apt/lists/*

# --- Python deps (non-torch) ---
RUN pip install --upgrade pip setuptools wheel \
 && pip install \
    numpy pandas scikit-learn networkx \
    matplotlib \
    yfinance alpha_vantage \
    opencv-python pillow \
    pympler pynvml

# --- PyTorch Geometric (GPU wheels matching torch+cuda) ---
# For torch 2.9.0 and CUDA 12.8, use cu128 wheels.
RUN python - <<'PY'
import sys, subprocess, torch
torch_ver = torch.__version__.split('+')[0]  # e.g., 2.9.0
cuda_suffix = "cu128"                        # CUDA 12.8 -> cu128
wheel_url = f"https://data.pyg.org/whl/torch-{torch_ver}+{cuda_suffix}.html"
pkgs = ["pyg-lib","torch-scatter","torch-sparse","torch-cluster","torch-spline-conv","torch-geometric"]
subprocess.check_call([sys.executable, "-m", "pip", "install", "-f", wheel_url, *pkgs])
print("Installed PyG wheels from:", wheel_url)
PY

# --- Copy project ---
COPY . /app

# Run the GUI by default
CMD ["python", "main.py"]
